正式版 main.c 的顺序如下。工程已经包含完整 User/main.c，不需要再次手工合并。

#include "gd32f30x.h"
#include "gd32f30x_misc.h"
#include "gd32f30x_rcu.h"

#include "FreeRTOS.h"
#include "task.h"

#include "app_boot_control.h"
#include "freertos_app.h"
#include "system_tasks.h"

int main(void)
{
    BaseType_t create_result;

    app_boot_control_vector_init();

    system_tasks_reset_reason_capture();
    rcu_all_reset_flag_clear();

    nvic_priority_group_set(NVIC_PRIGROUP_PRE4_SUB0);

    create_result = system_tasks_create();
    if (create_result != pdPASS)
    {
        freertos_app_fatal(FREERTOS_APP_FAULT_TASK_CREATE);
    }

    vTaskStartScheduler();

    freertos_app_fatal(FREERTOS_APP_FAULT_SCHEDULER_RETURNED);

    return 0;
}

不要在 main.c 中重复调用 power_gpio_init()、orin_power_on()、
external_watchdog_init()、voltage_monitor_init()、rtc_calendar_init() 或
shutdown_request_init()。这些调用由 system_tasks.c 按已经验证的启动顺序执行。

不要从 ISR 直接执行 Flash 擦写、阻塞延时、Orin 上下电或软件复位。
ISR 只清中断、锁存最小状态并通知对应所有者任务。
